export const setupNotifications = (tasks: any[]) => { if (!('Notification' in window)) return; if (Notification.permission !== 'granted' && Notification.permission !== 'denied') { Notification.requestPermission(); } // A simple foreground check for upcoming deadlines (e.g. next 24 hours). // Ideally, this runs periodically while the app is open. const checkDeadlines = () => { if (Notification.permission !== 'granted') return; const now = new Date(); const tomorrow = new Date(now.getTime() + 24 * 60 * 60 * 1000); const upcomingTasks = tasks.filter(t => { if (t.completed) return false; const tDate = new Date(t.date); // Remind if deadline is between now and +24h return tDate > now && tDate < tomorrow; }); upcomingTasks.forEach(task => { // Check if we already notified for this task recently const notifiedKey = `notified_${task.id}`; const lastNotified = localStorage.getItem(notifiedKey); if (!lastNotified || (now.getTime() - parseInt(lastNotified, 10) > 12 * 60 * 60 * 1000)) { new Notification("期限のリマインダー", { body: `期限が近づいています: ${new Date(task.date).toLocaleDateString()}`, icon: '/icon.png' }); localStorage.setItem(notifiedKey, now.getTime().toString()); } }); }; // Check immediately and then every hour checkDeadlines(); setInterval(checkDeadlines, 60 * 60 * 1000); };
